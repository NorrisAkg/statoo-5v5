357 static String _formatMonthLabel(String? monthStr) {
358 if (monthStr == null || monthStr.isEmpty) return '';
359 final parts = monthStr.split('-');
360 if (parts.length == 2) {
361 final year = parts[0];
362 final monthIndex = int.tryParse(parts[1]);
363 if (monthIndex != null && monthIndex >= 1 && monthIndex <= 12) {
364 return '${_frenchMonths[monthIndex - 1]} $year';
365 }
366 }
367 return monthStr;
368 }

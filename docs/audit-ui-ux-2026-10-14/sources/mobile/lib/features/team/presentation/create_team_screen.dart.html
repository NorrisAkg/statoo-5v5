39 Future<void> _submit() async {
40 if (!_formKey.currentState!.validate()) return;
41
42 setState(() => _isLoading = true);
43
44 try {
45 final team = TeamInfo(
46 id: 'team_${DateTime.now().millisecondsSinceEpoch}',
47 name: _nameController.text.trim(),
48 format: _selectedFormat,
49 location: _locationController.text.trim().isNotEmpty
50 ? _locationController.text.trim()
51 : null,
52 createdAt: DateTime.now(),
53 );
54
55 await ref.read(teamInfoProvider.notifier).saveTeam(team);
56
57 if (mounted) {
58 context.go('/ajouter-joueurs-initiaux');
59 }
60 } finally {
61 if (mounted) {
62 setState(() => _isLoading = false);
63 }
64 }
65 }
66
67 Future<void> _restoreFromBackup() async {
68 try {
69 final payload = await ref.read(backupServiceProvider).pickBackupFile();
70 if (payload == null || !mounted) return;
71
72 setState(() => _isLoading = true);
73 final result = await ref.read(backupServiceProvider).restoreBackup(payload, merge: false);
74
75 await ref.read(hasTeamProvider.notifier).reloadFromDb();
76 await ref.read(activeTeamIdProvider.notifier).reloadFromDb();
77 await ref.read(minPlayersPerSessionProvider.notifier).reload();
78
79 ref
80 ..invalidate(teamInfoProvider)
81 ..invalidate(allTeamsProvider)
82 ..invalidate(playersListProvider)
83 ..invalidate(allSessionsProvider)
84 ..invalidate(sessionsListProvider)
85 ..invalidate(activeSessionIdProvider)
86 ..invalidate(activeDraftSessionProvider)
87 ..invalidate(groupTotalsProvider)
88 ..invalidate(standingsProvider)
89 ..invalidate(standingsPerMatchProvider)
90 ..invalidate(topScorersProvider)
91 ..invalidate(topAssistsProvider)
92 ..invalidate(topCardsProvider);
93
94 if (mounted) {
95 ScaffoldMessenger.of(context).showSnackBar(
96 SnackBar(
97 content: Text(
98 'Restauration réussie : ${result.importedPlayersCount} joueurs et ${result.importedSessionsCount} séances restaurés.',
99 ),
100 ),
101 );
102 context.go('/');
103 }
104 } on Object catch (e) {
105 if (mounted) {
106 ScaffoldMessenger.of(context).showSnackBar(
107 SnackBar(content: Text('Erreur lors de la restauration : $e')),
108 );
109 }
110 } finally {
111 if (mounted) {
112 setState(() => _isLoading = false);
113 }
114 }
115 }

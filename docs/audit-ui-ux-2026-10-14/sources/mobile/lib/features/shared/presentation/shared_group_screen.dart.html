333 String _summary(String domain, Map<String, dynamic> payload) {
334 final dateValue = domain == 'sport'
335 ? (payload['session'] as Map)['played_at']
336 : payload['played_at'];
337 final date = DateTime.tryParse('$dateValue');
338 final label = date == null
339 ? 'Date non renseignée'
340 : DateFormat("d MMM yyyy '·' HH'h'mm", 'fr_FR').format(date);
341 if (domain == 'sport') {
342 final session = payload['session'] as Map<String, dynamic>;
343 return "$label · ${session['score_a'] ?? '–'} – ${session['score_b'] ?? '–'}";
344 }
345 final lines = payload['lines'] as List<dynamic>;
346 final received = lines.fold<int>(
347 0,
348 (n, l) =>
349 n + ((l as Map)['contribution'] as int) + (l['penalty_paid'] as int),
350 );
351 return '$label · Versements membres : $received F';
352 }

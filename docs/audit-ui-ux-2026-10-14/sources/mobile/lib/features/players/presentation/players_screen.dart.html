328 Future<bool> _confirmDeletePlayer(Player player) async {
329 final confirmed = await showDialog<bool>(
330 context: context,
331 builder: (ctx) {
332 return AlertDialog(
333 backgroundColor: context.colors.background900,
334 title: Text('Supprimer ${player.name} ?'),
335 content: Text(
336 "Ce joueur n'a participé à aucune séance. Sa suppression sera immédiate et définitive.",
337 style: TextStyle(color: context.colors.foreground300),
338 ),
339 actions: [
340 TextButton(
341 onPressed: () => Navigator.of(ctx).pop(false),
342 child: Text('Annuler', style: TextStyle(color: context.colors.foreground400)),
343 ),
344 ElevatedButton(
345 onPressed: () => Navigator.of(ctx).pop(true),
346 style: ElevatedButton.styleFrom(
347 backgroundColor: context.colors.danger,
348 foregroundColor: Colors.white,
349 ),
350 child: const Text('Supprimer'),
351 ),
352 ],
353 );
354 },
355 );
356
357 if (confirmed == true) {
358 final success = await ref.read(playersListProvider.notifier).removePlayer(player.id);
359 if (mounted) {
360 if (success) {
361 ScaffoldMessenger.of(context).showSnackBar(
362 SnackBar(content: Text('Le joueur "${player.name}" a été supprimé.')),
363 );
364 return true;
365 } else {
366 ScaffoldMessenger.of(context).showSnackBar(
367 const SnackBar(
368 content: Text('Impossible de supprimer ce joueur : il a déjà participé à une séance.'),
369 ),
370 );
371 }
372 }
373 }
374 return false;

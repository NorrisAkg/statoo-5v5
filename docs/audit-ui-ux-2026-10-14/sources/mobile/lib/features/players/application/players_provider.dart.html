105 Future<bool> removePlayer(String id) async {
106 final teamId = _requireTeamId();
107 await ref.read(sharedServiceProvider).requireWrite(teamId, 'members');
108 final canDelete = await ref.read(playersRepositoryProvider).canDeletePlayer(id, teamId: teamId);
109 if (!canDelete) return false;
110 await ref.read(playersRepositoryProvider).deletePlayer(id, teamId: teamId);
111 final currentList = state.value;
112 if (currentList != null) {
113 state = AsyncData(
114 currentList.where((p) => p.id != id).toList(),
115 );
116 }
117 _invalidateAll(id);

47 @override
48 Future<List<Player>> getPlayers({String? teamId}) async {

51 final rows = await db.query(
52 'players',
53 where: 'team_id = ?',
54 whereArgs: [teamId],
55 orderBy: 'is_active DESC, name ASC',
56 );
57
58 return rows.map((row) {
59 return Player(
60 id: row['id']! as String,
61 teamId: row['team_id']! as String,
62 name: row['name']! as String,
63 nickname: row['nickname'] as String?,
64 avatarUrl: row['avatar_url'] as String?,

66 createdAt: DateTime.parse(row['created_at']! as String),

95 @override
96 Future<PlayerDetailStats> getPlayerStats(String id, {required String teamId}) async {
97 final db = await _appDatabase.database;
98 final rows = await db.rawQuery('''
99 SELECT
100 COUNT(CASE WHEN status = 'CLOSED' AND is_ranked = 1 THEN 1 END) AS matches_played,
101 COALESCE(SUM(is_win), 0) AS wins,
102 COALESCE(SUM(is_draw), 0) AS draws,
103 COALESCE(SUM(is_loss), 0) AS losses,
104 COALESCE(SUM(points), 0) AS total_points,
105 COALESCE(SUM(CASE WHEN status = 'CLOSED' THEN goals ELSE 0 END), 0) AS goals,
106 COALESCE(SUM(CASE WHEN status = 'CLOSED' THEN assists ELSE 0 END), 0) AS assists,
107 COALESCE(SUM(CASE WHEN status = 'CLOSED' THEN yellow_cards ELSE 0 END), 0) AS yellow_cards,
108 COALESCE(SUM(CASE WHEN status = 'CLOSED' THEN red_cards ELSE 0 END), 0) AS red_cards,
109 COUNT(CASE WHEN status = 'CLOSED' THEN 1 END) AS appearances
110 FROM v_player_match_results
111 WHERE player_id = ? AND team_id = ?
112 AND player_id IN (SELECT id FROM players WHERE team_id = ?)
113 ''', [id, teamId, teamId]);
114
115 if (rows.isEmpty) {
116 return const PlayerDetailStats.empty();
117 }
118 return PlayerDetailStats.fromMap(rows.first);
119 }

150 @override
151 Future<Player> createPlayer({
152 required String name,
153 String? nickname,
154 String? avatarUrl,
155 String? teamId,
156 }) async {
157 final db = await _appDatabase.database;
158 final now = DateTime.now();
159
160 if (teamId == null || teamId.trim().isEmpty) {
161 throw ArgumentError('teamId is required to create a player');
162 }
163 final resolvedTeamId = teamId.trim();
164
165 final player = Player(
166 id: 'p_${now.millisecondsSinceEpoch}',
167 teamId: resolvedTeamId,
168 name: name.trim(),
169 nickname: nickname?.trim().isNotEmpty == true ? nickname!.trim() : null,
170 avatarUrl: avatarUrl,
171 createdAt: now,
172 updatedAt: now,
173 );
174
175 await db.insert('players', {
176 'id': player.id,
177 'team_id': player.teamId,
178 'name': player.name,
179 'nickname': player.nickname,
180 'avatar_url': player.avatarUrl,
181 'is_active': 1,
182 'created_at': player.createdAt.toIso8601String(),
183 'updated_at': player.updatedAt.toIso8601String(),
184 });
185
186 return player;
187 }

227 @override
228 Future<bool> canDeletePlayer(String id, {required String teamId}) async {
229 final db = await _appDatabase.database;
230 final count = Sqflite.firstIntValue(await db.rawQuery(
231 'SELECT COUNT(*) FROM participations WHERE player_id = ?',
232 [id],
233 )) ?? 0;
234 if (count > 0) return false;
235 // Only a player of this group can be deleted.
236 final owned = Sqflite.firstIntValue(await db.rawQuery(
237 'SELECT COUNT(*) FROM players WHERE id = ? AND team_id = ?',
238 [id, teamId],
239 )) ?? 0;
240 return owned > 0;
241 }

259 @override
260 Future<bool> isNameTaken({
261 required String name,
262 required String teamId,
263 String? excludePlayerId,
264 }) async {
265 final db = await _appDatabase.database;
266 final trimmed = name.trim();
267 if (trimmed.isEmpty) return false;
268
269 final where = StringBuffer('team_id = ? AND name = ? COLLATE NOCASE');
270 final args = <Object>[teamId, trimmed];
271
272 if (excludePlayerId != null) {
273 where.write(' AND id != ?');
274 args.add(excludePlayerId);
275 }
276
277 final count = Sqflite.firstIntValue(
278 await db.rawQuery(
279 'SELECT COUNT(*) FROM players WHERE $where',
280 args,
281 ),
282 ) ?? 0;
283 return count > 0;
284 }

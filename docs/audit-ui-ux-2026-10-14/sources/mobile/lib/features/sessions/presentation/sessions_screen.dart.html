258 Widget _buildSessionsList(BuildContext context, List<SessionSummaryItem> sessions) {
259 final grouped = <String, List<SessionSummaryItem>>{};
260 for (final item in sessions) {
261 final key = item.monthYearGroup;
262 grouped.putIfAbsent(key, () => []).add(item);
263 }
264
265 return ListView.builder(
266 padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
267 itemCount: grouped.keys.length,
268 itemBuilder: (context, index) {
269 final monthKey = grouped.keys.elementAt(index);
270 final monthSessions = grouped[monthKey]!;
271
272 return Column(
273 crossAxisAlignment: CrossAxisAlignment.start,
274 children: [
275 Padding(
276 padding: const EdgeInsets.only(top: 8, bottom: 12),
277 child: Text(
278 monthKey,
279 style: TextStyle(
280 color: context.colors.foreground400,
281 fontSize: 12,
282 fontWeight: FontWeight.bold,
283 letterSpacing: 0.5,
284 ),
285 ),
286 ),
287 ...monthSessions.map((item) => _buildSessionCard(context, item)),
288 const SizedBox(height: 12),
289 ],
290 );
291 },
292 );
293 }

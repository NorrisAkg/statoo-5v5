448 /// Closes the session if no blocking rule applies. Returns the blockers found
449 /// (empty when closed), re-checked on fresh data rather than on what the screen showed.
450 Future<List<CloseBlocker>> closeSession(String sessionId) async {
451 final teamId = _ref.read(activeTeamIdProvider);
452 if (teamId == null) throw StateError('No active team selected');
453 await _ref
454 .read(sharedServiceProvider)
455 .requireWrite(teamId, 'sport', sessionId: sessionId);
456
457 final session = await _repo.getSession(sessionId, teamId: teamId);
458 if (session == null) {
459 throw SessionCloseException('Session $sessionId not found');
460 }
461 final participants = await _repo.getSessionParticipants(
462 sessionId,
463 teamId: teamId,
464 );
465
466 final isTournament = session.sessionType == SessionType.tournament3;
467 final teamAParts = participants.where(
468 (p) => p.participation.team == Team.a,
469 );
470 final teamBParts = participants.where(
471 (p) => p.participation.team == Team.b,
472 );
473 final teamCParts = participants.where(
474 (p) => p.participation.team == Team.c,
475 );
476
477 final blockers = closeBlockers(
478 session: session,
479 teamACount: teamAParts.length,
480 teamBCount: teamBParts.length,
481 teamCCount: isTournament ? teamCParts.length : null,
482 playerGoalsA: teamAParts.fold<int>(
483 0,
484 (s, p) => s + p.participation.goals,
485 ),
486 playerGoalsB: teamBParts.fold<int>(
487 0,
488 (s, p) => s + p.participation.goals,
489 ),
490 playerGoalsC: teamCParts.fold<int>(
491 0,
492 (s, p) => s + p.participation.goals,
493 ),
494 playerAssistsA: teamAParts.fold<int>(
495 0,
496 (s, p) => s + p.participation.assists,
497 ),
498 playerAssistsB: teamBParts.fold<int>(
499 0,
500 (s, p) => s + p.participation.assists,
501 ),
502 playerAssistsC: teamCParts.fold<int>(
503 0,
504 (s, p) => s + p.participation.assists,
505 ),
506 participations: participants.map((p) => p.participation),
507 );
508 if (blockers.isNotEmpty) return blockers;
509
510 if (isTournament) {
511 final tournamentResult = computeTournamentResult(
512 session.winsA,
513 session.winsB,
514 session.winsC,
515 );
516 if (tournamentResult != null && tournamentResult != session.result) {
517 await _repo.updateTournamentWins(
518 sessionId: sessionId,
519 teamId: teamId,
520 winsA: session.winsA,
521 winsB: session.winsB,
522 winsC: session.winsC,
523 result: tournamentResult,
524 );
525 }
526 }
527
528 final closed = await _repo.closeSession(sessionId, teamId: teamId);
529 if (!closed) {
530 throw SessionCloseException('Session $sessionId could not be closed');
531 }
532 await _ref.read(sharedServiceProvider).queueSport(teamId, sessionId);
533 _invalidateAll(sessionId);

539 required String participationId,

541 Team? targetTeam,

555
556 final currentA = participants
557 .where((p) => p.participation.team == Team.a)
558 .length;
559 final currentB = participants
560 .where((p) => p.participation.team == Team.b)
561 .length;
562 final currentC = participants
563 .where((p) => p.participation.team == Team.c)
564 .length;
565 final isTournament = session.sessionType == SessionType.tournament3;
566
567 final newTeam =
568 targetTeam ??
569 (isTournament
570 ? (currentTeam == Team.a
571 ? Team.b
572 : (currentTeam == Team.b ? Team.c : Team.a))
573 : (currentTeam == Team.a ? Team.b : Team.a));
574
575 final nextA = newTeam == Team.a
576 ? (currentTeam == Team.a ? currentA : currentA + 1)
577 : (currentTeam == Team.a ? currentA - 1 : currentA);
578 final nextB = newTeam == Team.b
579 ? (currentTeam == Team.b ? currentB : currentB + 1)
580 : (currentTeam == Team.b ? currentB - 1 : currentB);
581 final nextC = newTeam == Team.c
582 ? (currentTeam == Team.c ? currentC : currentC + 1)
583 : (currentTeam == Team.c ? currentC - 1 : currentC);
584
585 if (!session.allowUnevenTeams) {
586 if (isTournament) {
587 final counts = [nextA, nextB, nextC];
588 final maxCount = counts.reduce((a, b) => a > b ? a : b);
589 final minCount = counts.reduce((a, b) => a < b ? a : b);
590 if (maxCount - minCount > 1) {
591 throw StateError(
592 "Ce changement créerait un écart d'effectif de ${maxCount - minCount} joueurs sans l'option équipes inégales.",
593 );
594 }
595 } else {
596 if ((nextA - nextB).abs() > 1) {
597 throw StateError(
598 "Ce changement créerait un écart d'effectif de ${(nextA - nextB).abs()} joueurs ($nextA vs $nextB) sans l'option équipes inégales.",
599 );
600 }

605 sessionId: sessionId,
606 participationId: participationId,
607 newTeam: newTeam,
608 teamId: teamId,
609 );

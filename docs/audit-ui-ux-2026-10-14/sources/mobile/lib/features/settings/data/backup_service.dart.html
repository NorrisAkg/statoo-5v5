23 /// Exporte l'intégralité des données en un fichier JSON et ouvre la feuille de partage native.
24 Future<BackupPayload> exportAndShareBackup() async {
25 final payload = await createBackupPayload();
26
27 final jsonString = const JsonEncoder.withIndent(' ').convert(payload.toJson());
28 final tempDir = await getTemporaryDirectory();
29 final timestamp = DateTime.now()
30 .toIso8601String()
31 .replaceAll(':', '-')
32 .split('.')
33 .first;
34 final file = File('${tempDir.path}/statoo_backup_$timestamp.json');
35 await file.writeAsString(jsonString);
36
37 await SharePlus.instance.share(
38 ShareParams(
39 files: [XFile(file.path, mimeType: 'application/json')],
40 subject: 'Sauvegarde Statoo 5v5',
41 text: 'Sauvegarde des données Statoo du ${DateTime.now().day}/${DateTime.now().month}/${DateTime.now().year}',
42 ),
43 );
44
45 return payload;
46 }

48 /// Construit l'objet BackupPayload à partir de la base SQLite et des SharedPreferences.
49 Future<BackupPayload> createBackupPayload() async {
50 final db = await _appDatabase.database;
51
52 final teams = await db.query('teams');
53 final team = teams.isNotEmpty ? teams.first : null;
54
55 final players = await db.query('players');
56 final sessions = await db.query('sessions');
57 final participations = await db.query('participations');
58 final metaRows = await db.query('meta');
59
60 final meta = <String, dynamic>{};
61 for (final row in metaRows) {
62 final key = row['key'] as String?;
63 if (key != null) {
64 meta[key] = row['value'];
65 }
66 }
67
68 final prefs = await SharedPreferences.getInstance();
69 final minPlayers = prefs.getInt('min_players_per_session');
70
71 return BackupPayload(
72 version: 2,
73 exportedAt: DateTime.now(),
74 team: team,
75 teams: teams,
76 players: players,
77 sessions: sessions,
78 participations: participations,
79 meta: meta,
80 minPlayersPerSession: minPlayers,
81 );
82 }

84 /// Ouvre le sélecteur de fichier et parse le fichier JSON sélectionné.
85 Future<BackupPayload?> pickBackupFile() async {
86 final files = await FilePicker.pickFiles();
87
88 if (files.isEmpty) {
89 return null;
90 }
91
92 final file = files.first;
93 final String content;
94
95 try {
96 content = await file.xFile.readAsString();
97 } on Object {
98 throw const FormatException('Impossible de lire le fichier sélectionné.');
99 }
100
101 final dynamic decoded;
102 try {
103 decoded = jsonDecode(content);
104 } on Object {
105 throw const FormatException('Le fichier sélectionné ne contient pas de JSON valide.');
106 }
107
108 if (decoded is! Map<String, dynamic>) {
109 throw const FormatException("Le fichier sélectionné n'est pas une sauvegarde Statoo valide.");
110 }
111
112 return BackupPayload.fromJson(decoded);
113 }

115 /// Restaure les données :
116 /// - Si `merge == false` : Écrasement total de la base et remplacement par la sauvegarde.
117 /// - Si `merge == true` : Fusion intelligente sans écraser les données locales.
118 Future<RestoreResult> restoreBackup(
119 BackupPayload payload, {
120 required bool merge,
121 String? targetTeamId,
122 }) async {
123 if (merge && payload.meta.keys.any((key) => key.startsWith('finance_') || key.startsWith('shared_') || key.startsWith('outbox_'))) {
124 throw const FormatException('Cette sauvegarde contient des finances ou un groupe partagé. Utilisez la restauration complète pour conserver leurs identifiants et leur historique.');
125 }
126 final db = await _appDatabase.database;
127
128 return await db.transaction((txn) async {
129 if (!merge) {
130 // --- MODE ÉCRASEMENT ---
131 // 1. Suppression dans l'ordre strict des clés étrangères (enfants d'abord)
132 await txn.delete('participations');
133 await txn.delete('sessions');
134 await txn.delete('players');
135 await txn.delete('meta');
136 await txn.delete('teams');
137
138 // 2. Insertion dans l'ordre strict des clés étrangères (parents d'abord)
139 final teamsToInsert = payload.teams.isNotEmpty
140 ? payload.teams
141 : (payload.team != null ? [payload.team!] : <Map<String, dynamic>>[]);
142
143 for (final t in teamsToInsert) {
144 await txn.insert('teams', t, conflictAlgorithm: ConflictAlgorithm.replace);
145 }
146
147 final defaultTeamId = teamsToInsert.isNotEmpty ? teamsToInsert.first['id'] as String : 'team_1';
148 final validTeamIds = teamsToInsert.map((t) => t['id'] as String).toSet();
149
150 for (final p in payload.players) {
151 final rawTeamId = p['team_id'] as String?;
152 final effectiveTeamId = (rawTeamId != null && validTeamIds.contains(rawTeamId))
153 ? rawTeamId
154 : defaultTeamId;
155 final playerRow = {...p, 'team_id': effectiveTeamId};
156 await txn.insert('players', playerRow, conflictAlgorithm: ConflictAlgorithm.replace);
157 }
158
159 for (final s in payload.sessions) {
160 final rawTeamId = s['team_id'] as String?;
161 final effectiveTeamId = (rawTeamId != null && validTeamIds.contains(rawTeamId))
162 ? rawTeamId
163 : defaultTeamId;
164 final sessionRow = {..._withValidStatus(s), 'team_id': effectiveTeamId};
165 await txn.insert('sessions', sessionRow, conflictAlgorithm: ConflictAlgorithm.replace);
166 }
167
168 for (final pt in payload.participations) {
169 await txn.insert('participations', _withValidParticipation(pt), conflictAlgorithm: ConflictAlgorithm.replace);
170 }
171
172 for (final entry in payload.meta.entries) {
173 await txn.insert('meta', {
174 'key': entry.key,
175 'value': entry.value?.toString(),
176 }, conflictAlgorithm: ConflictAlgorithm.replace);
177 }
178
179 final prefs = await SharedPreferences.getInstance();
180 if (payload.minPlayersPerSession != null) {
181 await prefs.setInt('min_players_per_session', payload.minPlayersPerSession!);
182 }
183
184 if (teamsToInsert.isNotEmpty) {
185 final firstTeam = teamsToInsert.first;
186 final teamId = firstTeam['id'] as String;
187 final teamName = firstTeam['name'] as String? ?? 'Équipe';
188 await prefs.setString('active_team_id', teamId);
189 await prefs.setString('team_name', teamName);
190 }
191
192 return RestoreResult(
193 isMerge: false,
194 teamName: teamsToInsert.isNotEmpty ? (teamsToInsert.first['name'] as String? ?? 'Équipe') : 'Équipe',
195 importedPlayersCount: payload.players.length,
196 importedSessionsCount: payload.sessions.length,
197 importedParticipationsCount: payload.participations.length,
198 );
199 } else {
200 // --- MODE FUSION INTELLIGENTE ---
201 final localTeams = await txn.query('teams');
202 String localTeamId;
203 String localTeamName;
204
205 if (localTeams.isEmpty) {
206 if (payload.teams.isNotEmpty) {
207 for (final t in payload.teams) {
208 await txn.insert('teams', t, conflictAlgorithm: ConflictAlgorithm.replace);
209 }
210 localTeamId = payload.teams.first['id']! as String;
211 localTeamName = payload.teams.first['name'] as String? ?? 'Équipe';
212 } else if (payload.team != null) {
213 await txn.insert('teams', payload.team!);
214 localTeamId = payload.team!['id']! as String;
215 localTeamName = payload.team!['name'] as String? ?? 'Équipe';
216 } else {
217 localTeamId = 'team_1';
218 localTeamName = 'Mon Équipe';
219 await txn.insert('teams', {
220 'id': localTeamId,
221 'name': localTeamName,
222 'format': '5v5',
223 'created_at': DateTime.now().toIso8601String(),
224 });
225 }
226 } else {
227 final matched = targetTeamId != null
228 ? localTeams.where((t) => t['id'] == targetTeamId).firstOrNull
229 : null;
230 if (matched != null) {
231 localTeamId = matched['id']! as String;
232 localTeamName = matched['name'] as String? ?? 'Équipe';
233 } else {
234 localTeamId = localTeams.first['id']! as String;
235 localTeamName = localTeams.first['name'] as String? ?? 'Équipe';
236 }
237 }
238
239 // 1. Réconciliation des Joueurs
240 final localPlayers = await txn.query('players', where: 'team_id = ?', whereArgs: [localTeamId]);
241 final localPlayersByName = <String, String>{};
242 final localPlayerIds = (await txn.query('players')).map((p) => p['id']! as String).toSet();
243
244 for (final lp in localPlayers) {
245 final name = (lp['name'] as String?)?.trim().toLowerCase();
246 final id = lp['id']! as String;
247 if (name != null && name.isNotEmpty) {
248 localPlayersByName[name] = id;
249 }
250 }
251
252 final playerMapping = <String, String>{};
253 var importedPlayersCount = 0;
254 var reusedPlayersCount = 0;
255
256 for (final bp in payload.players) {
257 final bpId = bp['id']! as String;
258 final bpName = (bp['name'] as String?)?.trim() ?? '';
259 final normalizedName = bpName.toLowerCase();
260
261 if (localPlayersByName.containsKey(normalizedName)) {
262 final matchedId = localPlayersByName[normalizedName]!;
263 playerMapping[bpId] = matchedId;
264 reusedPlayersCount++;
265 } else {
266 var targetId = bpId;
267 if (localPlayerIds.contains(targetId)) {
268 targetId = 'p_${DateTime.now().millisecondsSinceEpoch}_$importedPlayersCount';
269 }
270
271 final newPlayer = Map<String, dynamic>.from(bp);
272 newPlayer['id'] = targetId;
273 newPlayer['team_id'] = localTeamId;
274
275 await txn.insert('players', newPlayer);
276 localPlayerIds.add(targetId);
277 localPlayersByName[normalizedName] = targetId;
278 playerMapping[bpId] = targetId;
279 importedPlayersCount++;
280 }
281 }
282
283 // 2. Réconciliation des Séances (remapées vers localTeamId pour garantir l'intégrité référentielle)
284 final localSessions = await txn.query('sessions', where: 'team_id = ?', whereArgs: [localTeamId]);
285 final existingSessionIds = (await txn.query('sessions')).map((s) => s['id']! as String).toSet();
286 final existingPlayedAt = localSessions
287 .map((s) => s['played_at'] as String?)
288 .whereType<String>()
289 .toSet();
290
291 var importedSessionsCount = 0;
292 var skippedSessionsCount = 0;
293 var importedParticipationsCount = 0;
294
295 for (final bs in payload.sessions) {
296 final bsId = bs['id']! as String;
297 final bsPlayedAt = bs['played_at'] as String?;
298
299 if (existingSessionIds.contains(bsId) ||
300 (bsPlayedAt != null && existingPlayedAt.contains(bsPlayedAt))) {
301 skippedSessionsCount++;
302 continue;
303 }
304
305 final newSession = Map<String, dynamic>.from(_withValidStatus(bs));
306 newSession['team_id'] = localTeamId;
307
308 await txn.insert('sessions', newSession);
309 existingSessionIds.add(bsId);
310 if (bsPlayedAt != null) existingPlayedAt.add(bsPlayedAt);
311 importedSessionsCount++;
312
313 final relatedParticipations = payload.participations
314 .where((pt) => pt['session_id'] == bsId);
315
316 for (final pt in relatedParticipations) {
317 final originalPlayerId = pt['player_id'] as String?;
318 if (originalPlayerId == null) continue;
319
320 final mappedPlayerId = playerMapping[originalPlayerId];
321 if (mappedPlayerId != null) {
322 final newPt = Map<String, dynamic>.from(pt);
323 newPt['player_id'] = mappedPlayerId;
324
325 await txn.insert(
326 'participations',
327 _withValidParticipation(newPt),
328 conflictAlgorithm: ConflictAlgorithm.replace,
329 );
330 importedParticipationsCount++;
331 }
332 }
333 }
334
335 return RestoreResult(
336 isMerge: true,
337 teamName: localTeamName,
338 importedPlayersCount: importedPlayersCount,
339 importedSessionsCount: importedSessionsCount,
340 importedParticipationsCount: importedParticipationsCount,
341 reusedPlayersCount: reusedPlayersCount,
342 skippedSessionsCount: skippedSessionsCount,
343 );
344 }
345 });
346 }

348 Map<String, dynamic> _withValidParticipation(Map<String, dynamic> pt) {
349 var team = pt['team'];
350 if (team == 'a') team = 'A';
351 if (team == 'b') team = 'B';
352 return team != null ? {...pt, 'team': team} : pt;
353 }
354
355 /// Same rule as the v5 migration: a backup taken before it may still hold CLOSED
356 /// sessions without score, which would count as played matches worth 0 pts (CDC §4.2).
357 Map<String, dynamic> _withValidStatus(Map<String, dynamic> session) {
358 var res = session['result'];
359 if (res == 'team_a') res = 'A';
360 if (res == 'team_b') res = 'B';
361 if (res == 'draw') res = 'DRAW';
362
363 final normalized = Map<String, dynamic>.from(session);
364 if (res != null) {
365 normalized['result'] = res;
366 }
367 final unscored = normalized['score_a'] == null || normalized['score_b'] == null || normalized['result'] == null;
368 return normalized['status'] == 'CLOSED' && unscored ? {...normalized, 'status': 'DRAFT'} : normalized;
369 }

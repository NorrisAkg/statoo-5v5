48 Future<void> _load() async {
49 try {
50 final group = ref.read(activeTeamIdProvider);
51 if (group == null) throw StateError('Aucun groupe sélectionné.');
52 _group = group;
53 final session = await ref.read(
54 sessionDetailProvider(widget.sessionId).future,
55 );
56 final repo = ref.read(financeRepositoryProvider);
57 final storedPoint = await repo.getPoint(group, widget.sessionId);
58 if (session == null && storedPoint == null) {
59 throw StateError('Séance ou échéance introuvable.');
60 }
61 _savedBefore = await repo.isValidated(group, widget.sessionId);
62 final point = await repo.prepare(
63 group,
64 widget.sessionId,
65 session?.playedAt ?? storedPoint!.playedAt,
66 );
67 if (!mounted) return;
68 _point = point;
69 _cancelled = point.cancelled;
70 for (final line in point.lines) {
71 _contributions[line.memberId] = TextEditingController(
72 text: line.contribution == 0 ? '' : '${line.contribution}',
73 );
74 _penalties[line.memberId] = TextEditingController(
75 text: line.penaltyPaid == 0 ? '' : '${line.penaltyPaid}',
76 );
77 _due[line.memberId] = line.penaltyDue;
78 _reasons[line.memberId] = line.reason;
79 }
80 _expenses.addAll(point.expenses);
81 _donations.addAll(point.donations);
82 setState(() => _loaded = true);
83 } on Object catch (error) {
84 if (mounted) setState(() => _error = '$error');
85 }
86 }

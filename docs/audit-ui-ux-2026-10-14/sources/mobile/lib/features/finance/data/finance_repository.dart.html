58 Future<String> canonical(
59 String group,
60 String session, {
61 DateTime? date,
62 }) async {
63 final db = await database.database;
64 final occurrenceRows = await db.query(
65 'meta',
66 where: 'key = ?',
67 whereArgs: ['shared_occurrence_$session'],
68 );
69 String? occurrence = occurrenceRows.isEmpty
70 ? null
71 : occurrenceRows.single['value']! as String;
72 if (occurrence == null) {
73 final sessions = await db.query(
74 'sessions',
75 where: 'id = ? AND team_id = ?',
76 whereArgs: [session, group],
77 );
78 final playedAt =
79 date?.toIso8601String() ??
80 (sessions.isEmpty ? null : sessions.single['played_at'] as String?);
81 if (playedAt == null) return session;
82 occurrence = '${playedAt.substring(0, 10)}-1';
83 await db.insert('meta', {
84 'key': 'shared_occurrence_$session',
85 'value': occurrence,
86 }, conflictAlgorithm: ConflictAlgorithm.ignore);
87 }
88 final alias = await db.query(
89 'meta',
90 where: 'key = ?',
91 whereArgs: ['shared_finance_alias_${group}_$occurrence'],
92 );
93 if (alias.isNotEmpty) return alias.single['value']! as String;
94 await db.insert('meta', {
95 'key': 'shared_finance_alias_${group}_$occurrence',
96 'value': session,
97 }, conflictAlgorithm: ConflictAlgorithm.ignore);
98 return session;

126 Future<void> savePoint(
127 String group,
128 String session,
129 FinancialPoint point, {
130 String reason = '',
131 }) async {
132 point.validate();
133 final id = await canonical(group, session, date: point.playedAt);
134 await shared.requireWrite(group, 'finance', sessionId: id);
135 if (await isValidated(group, id) && reason.trim().isEmpty) {
136 throw const FormatException('Indiquez le motif de correction.');
137 }
138 final db = await database.database;
139 final known = (await db.query(
140 'players',
141 where: 'team_id = ?',
142 whereArgs: [group],
143 )).map((p) => p['id']).toSet();
144 if (point.lines.any((l) => !known.contains(l.memberId))) {
145 throw const FormatException('Membre inconnu dans ce groupe.');
146 }
147 final previous = await db.query(
148 'meta',
149 where: 'substr(key,1,?) = ? AND key != ?',
150 whereArgs: [
151 'finance_point_${group}_'.length,
152 'finance_point_${group}_',
153 key(group, id),
154 ],
155 );
156 final otherPoints = previous.map(
157 (p) => FinancialPoint.fromJson(
158 jsonDecode(p['value']! as String) as Map<String, dynamic>,
159 ),
160 );
161 final opening = group == shared.groupId
162 ? (shared.snapshot?['opening'] as Map?)
163 : null;
164 final balances = FinancialTotals.calculate(
165 [...otherPoints, point],
166 openingCash: opening?['cash'] as int? ?? 0,
167 openingDate: DateTime.tryParse(opening?['start_date'] as String? ?? ''),
168 openingMembers: {
169 for (final e in (opening?['members'] as Map? ?? {}).entries)
170 e.key as String: MemberBalance(
171 contributionBalance:
172 (e.value as Map)['contribution_balance'] as int,
173 penaltyDebt: (e.value as Map)['penalty_debt'] as int,
174 ),
175 },
176 );
177 if (balances.members.values.any((b) => b.penaltyDebt < 0)) {
178 throw const FormatException(
179 'Le paiement dépasse les pénalités dues. Enregistrez la pénalité due ou corrigez le paiement.',
180 );
181 }
182 await db.transaction((txn) async {
183 final old = await txn.query(
184 'meta',
185 where: 'key = ?',
186 whereArgs: [key(group, id)],
187 );
188 await txn.insert('meta', {
189 'key': 'finance_history_${group}_${shared.operationId()}',
190 'value': jsonEncode({
191 'session': id,
192 'reason': reason,
193 'previous': old.isEmpty ? null : old.single['value'],
194 'actor': shared.accountId,
195 'at': DateTime.now().toIso8601String(),
196 }),
197 });
198 await txn.insert('meta', {
199 'key': 'finance_validated_${group}_$id',
200 'value': '1',
201 }, conflictAlgorithm: ConflictAlgorithm.replace);
202 await txn.insert('meta', {
203 'key': key(group, id),
204 'value': jsonEncode(point.toJson()),
205 }, conflictAlgorithm: ConflictAlgorithm.replace);
206 await shared.enqueue(
207 txn,
208 group,
209 'finance',
210 id,
211 point.toJson(),
212 reason: reason,
213 );
214 });
215 }

235 Future<List<PendingFinancialPeriod>> pendingPeriods(
236 String group, {
237 required DateTime asOf,
238 DateTime? since,
239 }) async {
240 final db = await database.database;
241 final known = await periods(group);
242 final sessions = await db.query(
243 'sessions',
244 where: 'team_id = ?',
245 whereArgs: [group],
246 orderBy: 'played_at DESC',
247 );
248 final end = DateTime(asOf.year, asOf.month, asOf.day + 1);
249 final seen = <String>{};
250 final pending = <PendingFinancialPeriod>[];
251 for (final session in sessions) {
252 final date = DateTime.parse(session['played_at']! as String);
253 if (!date.isBefore(end) || (since != null && date.isBefore(since)))
254 continue;
255 final id = session['id']! as String;
256 final refs = await db.query(
257 'meta',
258 where: 'key = ?',
259 whereArgs: ['shared_occurrence_$id'],
260 );
261 final occurrence = refs.isEmpty
262 ? '${date.toIso8601String().substring(0, 10)}-1'
263 : refs.single['value']! as String;
264 if (!seen.add(occurrence)) continue;
265 final aliases = await db.query(
266 'meta',
267 where: 'key = ?',
268 whereArgs: ['shared_finance_alias_${group}_$occurrence'],
269 );
270 final financialId = aliases.isEmpty
271 ? id
272 : aliases.single['value']! as String;
273 if (!known.containsKey(financialId))
274 pending.add(PendingFinancialPeriod(sessionId: id, date: date));
275 }
276 return pending;
277 }

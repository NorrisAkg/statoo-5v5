192 /// Defense in depth for multi-tenancy: a participation must link a player and
193 /// a session of the same group, otherwise its stats would leak across groups.
194 Future<void> _createTenantTriggers(Database db) async {
195 const guard = '''
196 SELECT RAISE(ABORT, 'participation links a player and a session of different groups')
197 WHERE (SELECT team_id FROM players WHERE id = NEW.player_id)
198 IS NOT (SELECT team_id FROM sessions WHERE id = NEW.session_id);
199 ''';
200 await db.execute('''
201 CREATE TRIGGER IF NOT EXISTS trg_participations_same_team_insert
202 BEFORE INSERT ON participations
203 BEGIN
204 $guard
205 END;
206 ''');
207 await db.execute('''
208 CREATE TRIGGER IF NOT EXISTS trg_participations_same_team_update
209 BEFORE UPDATE OF player_id, session_id ON participations
210 BEGIN
211 $guard
212 END;
213 ''');
214 }

307 CREATE VIEW IF NOT EXISTS v_player_match_results AS
308 SELECT
309 pt.id AS participation_id,
310 pt.session_id,
311 pt.player_id,
312 pt.team,
313 pt.goals,
314 pt.assists,
315 pt.yellow_cards,
316 pt.red_cards,
317 s.team_id,
318 s.is_ranked,
319 s.status,
320 CASE
321 WHEN s.status = 'CLOSED' AND s.is_ranked = 1 THEN
322 CASE
323 WHEN s.result = 'DRAW' THEN 1
324 WHEN (s.result = 'A' AND pt.team = 'A') OR (s.result = 'B' AND pt.team = 'B') OR (s.result = 'C' AND pt.team = 'C') THEN 3
325 WHEN s.result = 'DRAW_AB' AND (pt.team = 'A' OR pt.team = 'B') THEN COALESCE(s.draw_points, 1)
326 WHEN s.result = 'DRAW_BC' AND (pt.team = 'B' OR pt.team = 'C') THEN COALESCE(s.draw_points, 1)
327 WHEN s.result = 'DRAW_AC' AND (pt.team = 'A' OR pt.team = 'C') THEN COALESCE(s.draw_points, 1)
328 WHEN s.result = 'DRAW_ALL' THEN COALESCE(s.draw_points, 1)
329 ELSE 0
330 END
331 ELSE 0
332 END AS points,
333 CASE
334 WHEN s.status = 'CLOSED' AND s.is_ranked = 1 AND ((s.result = 'A' AND pt.team = 'A') OR (s.result = 'B' AND pt.team = 'B') OR (s.result = 'C' AND pt.team = 'C')) THEN 1
335 ELSE 0
336 END AS is_win,
337 CASE
338 WHEN s.status = 'CLOSED' AND s.is_ranked = 1 AND (
339 s.result = 'DRAW' OR
340 (s.result = 'DRAW_AB' AND (pt.team = 'A' OR pt.team = 'B')) OR
341 (s.result = 'DRAW_BC' AND (pt.team = 'B' OR pt.team = 'C')) OR
342 (s.result = 'DRAW_AC' AND (pt.team = 'A' OR pt.team = 'C')) OR
343 s.result = 'DRAW_ALL'
344 ) THEN 1
345 ELSE 0
346 END AS is_draw,

348 WHEN s.status = 'CLOSED' AND s.is_ranked = 1 AND (

572 COALESCE(pc.active_players, 0) AS active_players,
573 COALESCE(sc.total_score_goals, 0) AS total_goals,
574 ROUND(CASE WHEN COALESCE(sc.total_sessions, 0) > 0 THEN CAST(COALESCE(sc.total_score_goals, 0) AS REAL) / sc.total_sessions ELSE 0.0 END, 2) AS avg_goals_per_session,
575 COALESCE(pt.total_assists, 0) AS total_assists,

577 COALESCE(pt.total_red_cards, 0) AS total_red_cards,
